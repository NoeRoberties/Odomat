\begin{frame}{C04.2 -- Analyse financi\`ere et benchmarks}
\textbf{Rep\`eres march\'e (source: \textit{Etude\_de\_march\_globale\_Odomat.pdf})}
\begin{itemize}
    \item Prix observ\'es pour RPG pixel art isom\'etriques PC: \textbf{15\$ \`a 30\$}.
    \item March\'e RPG mondial: \textbf{27,5 Md\$ (2024)} \textrightarrow{} \textbf{33,27 Md\$ (2026)}.
    \item Croissance moyenne: \textbf{8\% \`a 10\%} (TCAC), dynamique favorable mais concurrence forte.
\end{itemize}

\textbf{Benchmarks concurrents (ordre de grandeur)}
\begin{itemize}
    \item Hades: >1 million d'exemplaires vendus rapidement, >100 M\$ cumul\'es.
    \item Underrail: >100 000 ventes, suite financ\'ee via Kickstarter (500 k\$).
    \item Segment survivors/roguelike: forte traction (ex. Vampire Survivors, Risk of Rain 2).
\end{itemize}

\textbf{Lecture pour Odomat}
\begin{itemize}
    \item Positionnement prix cible coh\'erent pour un indie: \textbf{milieu-bas de fourchette} (acc\`es, d\'ecouverte).
    \item Diff\'erenciation produit issue du projet: combat bas\'e modules + progression \`a effets imm\'ediats.
    \item D\'ecision recommand\'ee: valider la traction via prototype jouable puis ajuster pricing selon retours joueurs.
\end{itemize}
\end{frame}
